const SC = { world: "w", source: "s", reserve: "r", quantity_unit: "u", mechanism: "m", basis: "b" };

function build({ retained, latest, linked }) {
  const launches = [];
  let store = retained;
  const d = {
    launch: async (command, _opts, authority, delayed) => {
      launches.push({ command, client: authority.client, grant: authority.grant, delayed: Boolean(delayed) });
      return { ok: false }; // every phase is "refused" so the chain stops after logging which authority it carried
    },
    ...(linked ? { linked } : {}),
    read: async () => ({ kind: "OPEN", time: {}, attempts: 1, latest }),
    scenario: () => SC,
    exitCode: () => 0,
    decline: () => {},
    closing: () => false,
    track: (p) => p,
    retain: (g) => { store = { kind: "RECORD", goal: { ...g } }; return true; },
    retained: () => store,
    pollMs: 1,
  };
  return { co: createGoalCoordinator(d), launches, get store() { return store; } };
}

const A = { scopes: ["player"], client: "clientA", grant: "grantA" };
const B = { scopes: ["player"], client: "clientB", grant: "grantB" };
const inflight = (act, amount) => ({ phase: "IN_FLIGHT", act_id: act, quantity: { amount, unit: "u" }, completes_at: {} });

async function run(name, cfg, request, authority, reqId, delayed) {
  const h = build(cfg);
  const r = await h.co.begin(request, authority, reqId, delayed);
  await h.co.settled();
  const g = h.store.kind === "RECORD" ? h.store.goal : null;
  console.log(`\n[${name}]`);
  console.log("  begin ->", r.ok ? `ok resumed=${r.resumed}` : `refused: ${r.reason.slice(0, 70)}`);
  console.log("  launches ->", h.launches.length ? JSON.stringify(h.launches) : "none");
  console.log("  final record ->", g ? JSON.stringify({ id: g.id, amount: g.amount, until: g.until, outcome: g.outcome, returned: g.returned, act_id: g.act_id, dispatch_ref: g.dispatch_ref }) : "none");
}
