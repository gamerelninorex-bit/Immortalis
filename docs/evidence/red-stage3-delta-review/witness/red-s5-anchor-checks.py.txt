import hashlib
import re
import sys
from pathlib import Path


def lines_of(path):
    return Path(path).read_text(encoding="utf-8").split("\n")


def show(path, pattern, ctx=0, flags=0):
    ls = lines_of(path)
    hit = False
    for i, line in enumerate(ls, 1):
        if re.search(pattern, line, flags):
            hit = True
            for j in range(max(1, i - ctx), min(len(ls), i + ctx) + 1):
                print(f"  {path}:{j}: {ls[j - 1].strip()[:170]}")
            if ctx:
                print("  --")
    if not hit:
        print(f"  {path}: NO MATCH for {pattern!r}")


def canon(t):
    return re.sub(r'<mention url="([^"]+)">[^<]*</mention>', r'<mention url="\1" />', t).strip()
